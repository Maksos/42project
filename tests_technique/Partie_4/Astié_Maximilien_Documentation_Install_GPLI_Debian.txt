||||||||||||||||GLPI||||||||||||||||

Connection via ssh sur la VM


******Mettre a jour le système******


apt update && apt full-upgrade -y


******Installation du serveur LAMP******

apt install apache2 mariadb-server php -y
systemctl enable apache2 mariadb

******Installation de GLPI******

------Dépendances de GLPI-------

apt install perl -y
apt install php-ldap php-imap php-apcu php-xmlrpc php-cas php-mysqli php-mbstring php-curl php-gd php-simplexml php-xml php-intl php-zip php-bz2 -y

--------------------------------

systemctl reload apache2


-------Télécharger le code de GLPI------

cd /tmp

wget https://github.com/glpi-project/glpi/releases/download/10.0.18/glpi-10.0.18.tgz

tar xzf glpi-10.0.18.tgz -C /var/www/html

----------------------------------------

---Permission au serveur Apache sur le dossier glpi---

chown -R www-data:www-data /var/www/html/glpi
chmod -R 755 /var/www/html/glpi

-------------------------------------------------------

---------Database-------------

mysql -u root

create database glpi;
create user glpiuser@localhost identified by 'strongpass';
grant all privileges on glpi.* to glpiuser@localhost;
flush privileges;
exit;


--------------------------
